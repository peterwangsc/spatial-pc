cmake_minimum_required(VERSION 3.22)
project(SpatialPakeWindows LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Spatial PC requires Windows x64")
endif()
set(BORINGSSL_SOURCE "" CACHE PATH "Clean pinned BoringSSL checkout")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${BORINGSSL_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${BORINGSSL_SOURCE}" status --porcelain
  OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../shared/pairing/boringssl.lock.json" lock)
string(JSON expected GET "${lock}" commit)
if(NOT revision STREQUAL expected OR dirty)
  message(FATAL_ERROR "BoringSSL must match the clean shared source lock")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(OPENSSL_NO_ASM ON CACHE BOOL "Portable shared PAKE configuration" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Private static crypto" FORCE)
set(BUILD_TESTING ON CACHE BOOL "Upstream primitive tests" FORCE)
set(BORINGSSL_PREFIX SPATIALPC_BSSL CACHE STRING "Private native symbols" FORCE)
add_subdirectory("${BORINGSSL_SOURCE}" boringssl EXCLUDE_FROM_ALL)
add_library(spatial_pake SHARED ../../shared/pairing/spatial_pake.c)
target_compile_definitions(spatial_pake PRIVATE SPATIAL_PAKE_SHARED BORINGSSL_PREFIX=SPATIALPC_BSSL OPENSSL_NO_ASM)
target_link_libraries(spatial_pake PRIVATE crypto)
set_target_properties(spatial_pake PROPERTIES LINKER_LANGUAGE CXX)
target_link_options(spatial_pake PRIVATE /OPT:REF /OPT:ICF)
